# Pinned to what the app was built and verified against (19 Aug 2026).
#
# These were ranges (`streamlit>=1.36.0`), which is fine on a machine where
# the installed version stays put and wrong for a deploy: Streamlit Cloud
# installs the newest release each time it builds, so the app would run on a
# version nobody has tried. This one leans on 1.38 behaviour in particular —
# fragment-scoped reruns, pixel column widths, and the fact that a toast does
# not survive `st.rerun()` (the whole `flash()` pattern exists because of it).
#
# Raising a pin is a deliberate change: bump it, run the tests, click through
# the pages, then commit. Not something a redeploy should decide on its own.
streamlit==1.38.0
pandas==2.0.3
# pandas 2.0.3 ships C extensions built against numpy 1.x. Left unpinned, a
# fresh install resolves numpy 2.x and pandas dies at import ("ValueError" in
# pandas._libs) — which is exactly how the first cloud deploy went down.
# 1.24.4 is what the app is verified against, and installs on 3.8 and 3.11.
numpy==1.24.4
gspread==6.2.1
google-auth==2.50.0
google-api-python-client==2.198.0
deep-translator==1.11.4
